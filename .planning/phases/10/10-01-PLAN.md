    // inlineStripCodeFences removes markdown triple-backtick fences from content.
    // This is a simplified version for task parsing, not tool call extraction.
    func inlineStripCodeFences(content string) string {
        var result strings.Builder
        lines := strings.Split(content, "\n")
        inFence := false
        for _, line := range lines {
            trimmed := strings.TrimSpace(line)
            if strings.HasPrefix(trimmed, "```") {
                inFence = !inFence
                continue
            }
            if !inFence {
                if result.Len() > 0 {
                    result.WriteString("\n")
                }
                result.WriteString(line)
            }
        }
        return result.String()
    }
    ```
