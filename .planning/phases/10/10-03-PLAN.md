    3a. `autoCompactAfterMessage(messages []m31types.Message) tea.Cmd`:
    ```go
    // autoCompactAfterMessage checks whether context consolidation should be triggered
    // after a completed message pair. If the consolidator is available and can consolidate,
    // returns a tea.Cmd that runs consolidation asynchronously.
    func (e *Engine) autoCompactAfterMessage(messages []m31types.Message) tea.Cmd {
        if e.consolidator == nil {
            return nil
        }

        // Sync the consolidator's message state (it needs to know current messages)
        e.consolidator.SetMessages(messages)

        // Check threshold: use token estimator if available, else rely on CanConsolidate()
        if e.tokens != nil {
            estTokens := e.tokens.EstimateMessages(messages)
            ctxLen := int64(types.DefaultContextLength)

            // Try to get actual context length from provider model info
            if e.provider != nil {
                if model, err := e.provider.GetModel(e.modelID); err == nil && model != nil {
                    ctxLen = model.ContextLength
                }
            }

            // Only auto-compact if over threshold AND CanConsolidate() returns true
            ratio := float64(estTokens) / float64(ctxLen)
            if ratio < m31types.AutoDreamThreshold {
                return nil
            }
        }

        // CanConsolidate is the final gate (checks paused, message count, protection rules)
        if !e.consolidator.CanConsolidate() {
            return nil
        }

        return e.autoCompactAsCmd()
    }
    ```

    3b. `autoCompactAsCmd() tea.Cmd`:
    ```go
    // autoCompactAsCmd returns a tea.Cmd that runs consolidation and emits a result.
    // This keeps consolidation in Bubble Tea's single-threaded model — the consolidation
    // itself is synchronous but runs via the tea.Cmd mechanism.
    func (e *Engine) autoCompactAsCmd() tea.Cmd {
        return func() tea.Msg {
            result := e.consolidator.Consolidate()
            if result == nil {
                return nil
            }
            e.logger.Info("auto-compaction triggered",
                "removed", result.MessagesRemoved,
                "tokens_saved", result.TokensSaved,
                "duration_ms", result.DurationMs,
            )
            return ConsolidationMsg{
                Result: result,
                Auto:   true,
            }
        }
    }
    ```
