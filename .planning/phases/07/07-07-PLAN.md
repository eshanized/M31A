    ```go
    type ModelItem struct {
        Model    types.ModelInfo
        Provider string
    }
    
    func (i ModelItem) Title() string {
        name := i.Model.Name
        if name == "" {
            name = i.Model.ID
        }
        return fmt.Sprintf("%s [%s]", name, providerBadge(i.Provider))
    }
    
    func (i ModelItem) Description() string {
        costDesc := fmt.Sprintf("$%.2f/M in · $%.2f/M out", 
            i.Model.Pricing.InputPerMToken, 
            i.Model.Pricing.OutputPerMToken)
        if i.Model.Pricing.InputPerMToken == 0 && i.Model.Pricing.OutputPerMToken == 0 {
            costDesc = "pricing N/A"
        }
        ctxDesc := fmt.Sprintf("context: %dK", i.Model.ContextLength/1024)
        caps := capabilityString(i.Model.Capabilities)
        return fmt.Sprintf("%s · %s · %s", costDesc, ctxDesc, caps)
    }
    
    func (i ModelItem) FilterValue() string {
        return strings.ToLower(i.Model.Name + " " + i.Model.ID)
    }
    ```

    ```go
    func fetchModelsCmd(registry *provider.ProviderRegistry) tea.Cmd {
        return func() tea.Msg {
            providers := registry.List()
            var allModels []types.ModelInfo
            for _, name := range providers {
                provider, err := registry.Get(name)
                if err != nil { continue }
                models, err := provider.FetchModels(context.Background())
                if err != nil { continue }
                for i := range models {
                    models[i].Provider = name  // tag with provider name
                }
                allModels = append(allModels, models...)
            }
            return modelFetchCompleteMsg{models: allModels}
        }
    }
    ```

    ```go
    func (m ModelSelector) filteredItems(items []list.Item) []list.Item {
        var result []list.Item
        for _, item := range items {
            mi := item.(ModelItem)
            // Provider filter
            if m.filter == filterOpenRouter && mi.Provider != "openrouter" { continue }
            if m.filter == filterZen && mi.Provider != "zen" { continue }
            // Search filter (handled by list's built-in filtering or manual)
            result = append(result, item)
        }
        return result
    }
    ```

    ```go
        b.WriteString(fmt.Sprintf("Capabilities: %s\n", capabilityString(d.Capabilities)))
        return lipgloss.NewStyle().Padding(1, 2).Border(lipgloss.RoundedBorder()).Render(b.String())
    }
    ```

    13. Helpers:
    ```go
    func providerBadge(provider string) string {
        switch provider {
        case "openrouter": return "OR" // in Brand color
        case "zen": return "ZEN"       // in Thinking color
        default: return provider
        }
    }
    
    func capabilityString(c CapFlags) string {
        var parts []string
        if c.Tools { parts = append(parts, "tools") }
        if c.Reasoning { parts = append(parts, "reasoning") }
        if c.Vision { parts = append(parts, "vision") }
        if len(parts) == 0 { return "basic" }
        return strings.Join(parts, ", ")
    }
    ```
