    ### Inline Editing

    For each field, when the field is focused and user presses Enter:
    1. Toggle `field.editing = true`
    2. If editing, further keypresses modify the value directly
    3. When editing is true, pressing Enter again toggles back and marks dirty
    4. Pressing Esc cancels edit (restores original value)

    For bool fields: Enter toggles true/false immediately (no edit mode).
    For int/float: validate input; reject non-numeric chars.
    For string: accept all chars.

    Each tab's view function renders fields: label then value. If editing, value is shown inline with cursor (using `strings.Builder` or simple `>` marker). If masked, show "••••••••" instead of actual value, with a "(press E to reveal)" hint.

    API key reveal: when a masked field is focused and editing, set `field.masked = false` temporarily to show the actual key. When edit mode exits (Enter or Esc), re-mask.

    ### View

    ```go
    func (m SettingsModel) View() string {
        if m.width == 0 { return "Loading..." }
    
        // Tab bar
        var tabBar strings.Builder
        for i := 0; i < int(tabCount); i++ {
            tab := settingsTab(i)
            name := tabNames[tab]
            if tab == m.activeTab {
                tabBar.WriteString(activeTabStyle.Render(name))
            } else {
                tabBar.WriteString(inactiveTabStyle.Render(name))
            }
            if i < int(tabCount)-1 {
                tabBar.WriteString("  ")
            }
        }
    
        // Field content for active tab
        content := m.renderActiveTab()
    
        // Footer with help
        footer := helpStyle.Render("[Tab/←→] Navigate  [Enter] Edit  [Esc] Back  [Ctrl+S] Save")
        if m.dirty {
            footer = dirtyStyle.Render("Unsaved changes • ") + footer
        }
        if m.err != "" {
            footer = errorStyle.Render(m.err) + "\n" + footer
        }
    
        return lipgloss.JoinVertical(lipgloss.Top, tabBar.String(), content, footer)
    }
    ```

    ### Update

    ```go
    func (m SettingsModel) Update(msg tea.Msg) (SettingsModel, tea.Cmd) {
        switch msg := msg.(type) {
        case tea.WindowSizeMsg:
            m.width = msg.Width
            m.height = msg.Height
        case tea.KeyMsg:
            switch msg.String() {
            case "ctrl+c", "esc":
                // If editing, cancel edit
                // Otherwise, return quit cmd
            case "tab", "right":
                if m.isEditing() {
                    // insert char or navigate within field
                } else {
                    m.nextTab()
                }
            case "shift+tab", "left":
                if !m.isEditing() {
                    m.prevTab()
                }
            case "enter":
                if m.isEditing() {
                    m.confirmEdit()
                } else {
                    m.startEdit()
                }
            case "ctrl+s":
                return m, m.saveCmd()
            case "backspace":
                if m.isEditing() {
                    m.deleteChar()
                }
            default:
                if m.isEditing() && len(msg.String()) == 1 {
                    m.insertChar(msg.String())
                }
            }
        }
        return m, nil
    }
    ```
