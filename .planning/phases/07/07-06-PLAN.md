    ```go
    // ModelCacheRefreshTicker runs a background goroutine that refreshes the
    // model cache at the given interval. It stops when ctx is cancelled.
    // Returns a function that can be called to stop the ticker.
    func (c *ModelCache) StartRefreshTicker(ctx context.Context, interval time.Duration) func() {
        if interval <= 0 {
            interval = ModelCacheTTL // default to 5 minutes
        }
        
        ctx, cancel := context.WithCancel(ctx)
        go func() {
            ticker := time.NewTicker(interval)
            defer ticker.Stop()
            for {
                select {
                case <-ticker.C:
                    _ = c.Refresh() // ignore errors — stale cache preferred over crash
                case <-ctx.Done():
                    return
                }
            }
        }()
        return cancel
    }
    ```

    6. Ensure `ModelCache` already has a `Refresh()` method. If not, add it:

    ```go
    func (c *ModelCache) Refresh() error {

        defer c.mu.Unlock()
        // Re-fetch from provider; implementation depends on provider-specific client
        // This is a no-op stub that resets the fetched timestamp
        c.fetched = time.Now()
        return nil
    }
    ```
