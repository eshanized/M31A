    In `internal/testutil/ci/ci_test.go`, replace `t.Parallel()` + manual `os.Setenv`/`os.Unsetenv` with `t.Setenv()` which auto-restores and is safe for parallel tests.

    **Current (racy):**
    ```go
    func TestIsCI(t *testing.T) {
        tests := []struct {
            name     string
            envVar   string
            envValue string
            want     bool
        }{
            {"GITHUB_ACTIONS", "GITHUB_ACTIONS", "true", true},
            {"GITLAB_CI", "GITLAB_CI", "true", true},
            // ... more cases
        }

        for _, tt := range tests {
            t.Run(tt.name, func(t *testing.T) {
                t.Parallel()  // REMOVE THIS
                old := os.Getenv(tt.envVar)
                os.Setenv(tt.envVar, tt.envValue)
                defer os.Setenv(tt.envVar, old)  // RACE: cleanup runs after parallel test completes
                // ...
                got := IsCI()
                if got != tt.want {
                    t.Errorf("IsCI() = %v, want %v", got, tt.want)
                }
            })
        }
    }
    ```

    **Fixed (using t.Setenv):**
    ```go
    func TestIsCI(t *testing.T) {
        tests := []struct {
            name     string
            envVar   string
            envValue string
            want     bool
        }{
            {"GITHUB_ACTIONS", "GITHUB_ACTIONS", "true", true},
            {"GITLAB_CI", "GITLAB_CI", "true", true},
            // ... more cases
        }

        for _, tt := range tests {
            t.Run(tt.name, func(t *testing.T) {
                t.Setenv(tt.envVar, tt.envValue)  // Auto-restores after subtest, safe with parallel
                // t.Parallel() — CAN keep if desired, t.Setenv handles it
                got := IsCI()
                if got != tt.want {
                    t.Errorf("IsCI() = %v, want %v", got, tt.want)
                }
            })
        }
    }
    ```

    Per D-11: Use `t.Setenv()` (Go 1.17+) which auto-restores environment variables. This is the modern, race-free pattern. Can keep `t.Parallel()` if desired since `t.Setenv` handles isolation.

    Per D-08: Only change the test. Do not modify IsCI() implementation in ci.go.
