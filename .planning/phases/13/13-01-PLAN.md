    1. In `internal/tui/commands.go`, add a `handleExport` function:
       ```go
       func handleExport(args []string, ctx CommandContext) CommandResult {
           if ctx.SessionManager == nil || ctx.SessionID == "" {
               return CommandResult{Success: false, Message: "No active session. Start a session first."}
           }

           // Parse flags
           format := "markdown" // --format markdown|html, default markdown
           outputPath := ""    // --output <file>, default stdout

           for i := 0; i < len(args); i++ {
               switch args[i] {
               case "--format":
                   if i+1 < len(args) {
                       i++
                       switch args[i] {
                       case "markdown", "html":
                           format = args[i]
                       default:
                           return CommandResult{Success: false, Message: fmt.Sprintf("Invalid format: %q. Use markdown or html.", args[i])}
                       }
                   }
               case "--output":
                   if i+1 < len(args) {
                       i++
                       outputPath = args[i]
                   }
               default:
                   return CommandResult{Success: false, Message: fmt.Sprintf("Unknown flag: %q. Usage: /export [--format markdown|html] [--output <file>]", args[i])}
               }
           }

           // Load session
           session, err := ctx.SessionManager.LoadSession(ctx.SessionID)
           if err != nil {
               return CommandResult{Success: false, Message: fmt.Sprintf("Failed to load session: %v", err)}
           }

           if outputPath == "" {
               // Print to stdout as command result message
               var output string
               switch format {
               case "markdown":
                   output, err = session.FormatSessionAsMarkdown(session)
               case "html":
                   output, err = session.FormatSessionAsHTML(session)
               }
               if err != nil {
                   return CommandResult{Success: false, Message: fmt.Sprintf("Export failed: %v", err)}
               }
               return CommandResult{Success: true, Message: output}
           }

           // Write to file
           err = session.ExportSessionToFile(session, format, outputPath)
           if err != nil {
               return CommandResult{Success: false, Message: fmt.Sprintf("Export to file failed: %v", err)}
           }

           // Also cache in session export directory
           baseDir := ctx.SessionManager.BaseDir()  // need to expose or use filepath.Dir
           exportDir := session.ExportDirForSession(baseDir, ctx.SessionID)
           cachedPath, _ := session.ExportSessionToDir(session, format, exportDir)

           msg := fmt.Sprintf("Session exported as %s to %s", format, outputPath)
           if cachedPath != "" {
               msg += fmt.Sprintf(" (cached at %s)", cachedPath)
           }
           return CommandResult{Success: true, Message: msg}
       }
       ```

| T-13-01-02 | Denial of Service | Large session export | mitigate | 5MB cap via MaxFileSize constant; pre-check estimate and post-build truncation prevent memory exhaustion |
| T-13-01-03 | Tampering | HTML injection in session content | mitigate | Message content is rendered inside `<pre>` blocks — no HTML interpretation. CSS is hardcoded, not user-controllable. |
| T-13-01-04 | Information Disclosure | Export file written to disk | accept | File written to session export/ subdirectory (same permissions as session data); user controls --output destination |
