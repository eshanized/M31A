# Plan 30-05: Screen-by-Screen Polish

## Objective

Polish all 8 major screens (Plan, Execute, Verify, Ship, Settings, ModelSelector, Discuss, GoalInput) with consistent visual hierarchy, opencode-inspired layouts, and compact information density.

## must_haves

- Plan screen: wave-grouped tasks, color-coded waves, dep arrows
- Execute screen: progress bar, live tool output, pause overlay
- Verify screen: pass/fail checklist, inline error detail, heal UI
- Ship screen: summary card with branded border, stats grid
- Settings screen: left-nav tabs, card-style content panels
- Model selector: two-pane list + detail view, provider filter pills
- Discuss screen: dot progress, card-styled question, compact timer
- Goal input: minimal, recent goals panel
- `CGO_ENABLED=0 go build` passes, tests pass

```go
func renderPlanHeader(t theme.Theme, taskCount int, cost float64, modelName, provider string) string {
    header := lipgloss.NewStyle().Foreground(t.Brand).Render("📋 Plan") +
        lipgloss.NewStyle().Foreground(t.TextSecondary).Render(fmt.Sprintf(" · %d tasks", taskCount))
    if cost > 0 {
        header += lipgloss.NewStyle().Foreground(t.TextMuted).Render(fmt.Sprintf(" · ~$%.4f", cost))
    }
    if modelName != "" {
        prov := providerShortName(provider)
        header += lipgloss.NewStyle().Foreground(t.TextMuted).Render(fmt.Sprintf(" · %s [%s]", modelName, prov))
    }
    return header
}
```

```go
func renderProgressBar(t theme.Theme, done, total int, width int) string {
    if total == 0 { return "" }
    pct := float64(done) / float64(total)
    barWidth := width - 2  // [ and ]
    filled := int(math.Round(pct * float64(barWidth)))
    bar := lipgloss.NewStyle().Foreground(t.Brand).Render(
        strings.Repeat(t.BlockFull, filled) +
        lipgloss.NewStyle().Foreground(t.Surface).Render(strings.Repeat(t.BlockLow, barWidth-filled)).String())
    return "[" + bar + "]"
}
```
