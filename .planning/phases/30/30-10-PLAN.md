<action>
In `internal/tui/theme/colors.go`, add color profile detection and fallback:

```go
// ColorProfile represents terminal color capability
type ColorProfile int

const (
    ProfileTrueColor ColorProfile = iota
    Profile256
    Profile16
)

// DetectColorProfile determines terminal color capability
func DetectColorProfile() ColorProfile {
    if lipgloss.HasDarkBackground() {
        // lipgloss determines profile from TERM/COLORTERM
        return ProfileTrueColor  // default
    }
    // Check COLORTERM env var
    ct := os.Getenv("COLORTERM")
    if ct == "truecolor" || ct == "24bit" {
        return ProfileTrueColor
    }
    term := os.Getenv("TERM")
    if strings.Contains(term, "256") {
        return Profile256
    }
    return Profile16
}

// ResolveColor maps a hex color to the best available terminal color
func ResolveColor(hex string, profile ColorProfile) lipgloss.Color {
    return lipgloss.Color(hex)  // lipgloss handles fallback internally
}
```

**Note**: Lipgloss handles color approximation automatically. The main change is ensuring we detect the profile and use it to decide which palette to load. For 16-color terminals, use ANSI named colors instead of hex.

Create a `PaletteProfile` function:
```go
func PaletteForProfile(profile ColorProfile) Theme {
    switch profile {
    case Profile16:
        return ansiPalette()
    default:
        return Default()
    }
}

func ansiPalette() Theme {
    t := Dark()  // start from dark
    // Override with ANSI colors
    t.Brand = lipgloss.Color("208")  // ANSI orange
    t.Success = lipgloss.Color("2")  // ANSI green
    t.Error = lipgloss.Color("1")    // ANSI red
    t.Warning = lipgloss.Color("3")  // ANSI yellow
    t.Thinking = lipgloss.Color("4") // ANSI blue
    applyThemeStyles(&t)
    return t
}
```

Wire into `ThemeManager`:
```go
func NewManager(mode Mode) *Manager {
    m := &Manager{mode: mode}
    m.profile = DetectColorProfile()
    m.resolve()
    return m
}
```

    // Full layout
