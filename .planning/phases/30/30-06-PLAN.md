type BadgeType int

const (
    BadgeBrand BadgeType = iota
    BadgeSuccess
    BadgeError
    BadgeWarning
    BadgeInfo
    BadgeNeutral
)

type Badge struct {
    Text    string
    Type    BadgeType
    Compact bool  // compact = no padding, for inline use
    Theme   theme.Theme
}

func (b Badge) Render() string {
    style := lipgloss.NewStyle().Bold(true)
    switch b.Type {
    case BadgeBrand:
        return style.Foreground(b.Theme.Brand).Render(b.Text)
    case BadgeSuccess:
        return style.Foreground(b.Theme.Success).Render("✓ " + b.Text)
    case BadgeError:
        return style.Foreground(b.Theme.Error).Render("✗ " + b.Text)
    case BadgeWarning:
        return style.Foreground(b.Theme.Warning).Render("⚠ " + b.Text)
    case BadgeInfo:
        return style.Foreground(b.Theme.Thinking).Render(b.Text)
    default:
        return style.Foreground(b.Theme.TextMuted).Render(b.Text)
    }
}
```

type SectionDivider struct {
    Title string
    Width int
    Theme theme.Theme
}

func (d SectionDivider) Render() string {
    if d.Width < 4 {
        return ""
    }
    if d.Title == "" {
        return lipgloss.NewStyle().
            Foreground(d.Theme.TextMuted).
            Faint(true).
            Render(strings.Repeat("─", d.Width))
    }
    // ── title ────────────────────────────
    label := " " + d.Title + " "
    remaining := d.Width - lipgloss.Width(label)
    if remaining <= 0 {
        return label
    }
    left := strings.Repeat("─", max(2, remaining/2-1))
    right := strings.Repeat("─", remaining-len(left)-1)
    return lipgloss.NewStyle().
        Foreground(d.Theme.TextMuted).
        Faint(true).
        Render("─" + left + label + right + "─")
}

func max(a, b int) int {
    if a > b { return a }
    return b
}
```

Create `internal/tui/components/card.go`:

```go
type CardStyle int

const (
    CardDefault CardStyle = iota
    CardBrand
    CardSuccess
    CardError
    CardWarning
)

type Card struct {
    Title   string
    Content string
    Width   int
    Border  lipgloss.Border  // theme.ThinBorder, theme.NormalBorder, theme.DoubleBorder
    Style   CardStyle
    Theme   theme.Theme
}

func (c Card) Render() string {
    if c.Border == (lipgloss.Border{}) {
        c.Border = lipgloss.RoundedBorder()
    }
    borderColor := c.Theme.Border
    switch c.Style {
    case CardBrand:
        borderColor = c.Theme.Brand
    case CardSuccess:
        borderColor = c.Theme.Success
    case CardError:
        borderColor = c.Theme.Error
    case CardWarning:
        borderColor = c.Theme.Warning
    }
    
    style := lipgloss.NewStyle().
        Border(c.Border).
        BorderForeground(borderColor).
        Width(c.Width).
        Padding(0, 1)
    
    if c.Title != "" {
        header := lipgloss.NewStyle().
            Foreground(borderColor).
            Bold(true).
            Render(c.Title)
        return style.Render(header + "\n" + c.Content)
    }
    return style.Render(c.Content)
}
```

type Toast struct {
    Text      string
    Type      string   // "success", "error", "warning", "info"
    CreatedAt time.Time
}

func renderToast(t theme.Theme, toast Toast, width int) string {
    var borderColor lipgloss.Color
    switch toast.Type {
    case "success": borderColor = t.Success
    case "error":   borderColor = t.Error
    case "warning": borderColor = t.Warning
    default:        borderColor = t.Brand
    }
    
    return lipgloss.NewStyle().
        Border(theme.ThinBorder).
        BorderForeground(borderColor).
        Padding(0, 1).
        Foreground(t.TextPrimary).
        Render(toast.Text)
}
