    ```go
    // streamLLMStreaming sends a chat request and returns the underlying
    // StreamIterator. The caller is responsible for iterating via Next()
    // and emitting each chunk to the TUI (typically via MsgEmitter).
    //
    // This is the streaming variant used by phases that need to render
    // tokens progressively (Discuss phase, future phases). For phases
    // that consume the entire response in one shot (Plan, Execute task
    // generation), use the synchronous streamLLM instead.
    func (e *Engine) streamLLMStreaming(ctx context.Context, messages []m31types.Message, toolsEnabled bool) (*types.StreamIterator, error) {
        if e.provider == nil {
            return nil, fmt.Errorf("no active provider")
        }
        req := e.buildChatRequest(messages, toolsEnabled, false)
        return e.provider.ChatCompletionStream(ctx, req)
    }
    ```

    In `internal/workflow/discuss.go`, replace `runDiscuss` to use the
    streaming variant. The new flow:

    1. Call `streamLLMStreaming` to get the iterator
    2. Iterate via `iterator.Next()` and emit each chunk to the TUI
       via `e.msgEmitter.Emit(StreamChunkMsg{Chunk: chunk, ...})`
    3. After the stream ends, accumulate the full content, parse
       questions, and return the PhaseResult as before

    ```go
    func (e *Engine) runDiscuss(ctx context.Context, goal string) (*PhaseResult, error) {
        e.logger.Info("discuss phase starting")

        messages := e.buildDiscussContext(goal)

        // Get the streaming iterator
        iterator, err := e.streamLLMStreaming(ctx, messages, false)
        if err != nil {
            return &PhaseResult{
                Phase: m31types.PhaseDiscuss, Success: false, Error: err.Error(),
            }, err
        }
        defer iterator.Close()

        // Accumulate content and stream chunks to the TUI
        var content strings.Builder
        for {
            chunk, err := iterator.Next()
            if err != nil {
                if err == io.EOF {
                    break
                }
                return &PhaseResult{
                    Phase: m31types.PhaseDiscuss, Success: false, Error: err.Error(),
                }, err
            }
            if chunk == nil {
                break
            }
            content.WriteString(chunk.Delta)

            // Emit the chunk to the TUI for progressive rendering
            if e.msgEmitter != nil {
                e.msgEmitter.Emit(StreamChunkMsg{
                    Chunk:  chunk,
                    Source: "discuss",
                })
            }
        }

        fullContent := content.String()
        questions := parseQuestions(fullContent)

        msg := m31types.Message{
            Role:    "assistant",
            Content: fullContent,
        }

        e.discussState = DiscussState{Questions: questions}

        return &PhaseResult{
            Phase:        m31types.PhaseDiscuss,
            Success:      true,
            Messages:     []m31types.Message{msg},
            NeedsAnswers: len(questions) > 0,
        }, nil
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Engine → TUI (D-07) | Engine emits StreamChunkMsg via MsgEmitter. TUI trusts chunk content is well-formed JSON from the provider. |
| TUI → User (D-07) | User sees streamed tokens; no security boundary. |
| TUI internal (D-08) | Setter methods enforce encapsulation but don't introduce a trust boundary. |
| TUI → Config (D-09) | Sidebar threshold is user-configurable in `~/.m31a/config.toml`. No security impact. |
| Streaming lifecycle (D-10) | `streamCh` close signals "no more data." No untrusted sender. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-05-01 | Tampering | Discuss phase streamed content | accept | Content is the LLM's own response; same trust as the existing Plan/Execute streaming. |
| T-14-05-02 | Denial of Service | User sets SidebarWidthThreshold to 0 | accept | Threshold 0 means sidebar never auto-shows; user must manually toggle. Acceptable. |
| T-14-05-03 | Race Condition | streamCh close before streamDone close | mitigate | Defer LIFO order ensures `close(streamCh)` runs first (data channel signals end), then `close(streamDone)` (goroutine signals end). |
| T-14-05-04 | Tampering | ModelSelector.SetRegistry with nil | accept | Nil registry is a valid state (model selector may be in initial state). Existing code already handles this. |
| T-14-05-05 | Information Disclosure | Streamed tokens in REPL | accept | Tokens are the user's own goal refinement; not sensitive. |
</threat_model>
