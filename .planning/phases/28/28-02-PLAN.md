Add to `internal/tui/helpers.go`:

```go
func (m *AppState) loadAndRestoreSession(sessionID string, clearExisting bool) tea.Cmd {
    session, err := session.Load(sessionID)
    if err != nil {
        return func() tea.Msg { return errorMsg{err} }
    }
    
    m.ensureReplModel()
    m.replModel.SetProvider(m.activeProvider)
    m.replModel.SetDispatcher(m.dispatcher)
    m.replModel.SetSessionID(sessionID)
    
    if clearExisting {
        m.replModel.ClearMessages()
        m.replModel.SetCommandRegistry(m.commandRegistry)
    }
    
    m.replModel.AddMessages(session.Messages)
    m.propagateSessionID(sessionID)
    
    return nil
}
```
