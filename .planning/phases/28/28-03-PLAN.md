func RenderSectionHeader(title string, width int) string {
    prefix := fmt.Sprintf("── %s ", title)
    remaining := width - lipgloss.Width(prefix)
    if remaining < 0 {
        remaining = 0
    }
    return prefix + strings.Repeat("─", remaining)
}
