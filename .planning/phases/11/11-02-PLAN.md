       ```go
       func Load(path string) (*Config, error) {
           // Step 1: M31A_CONFIG env var overrides path
           if envPath := os.Getenv("M31A_CONFIG"); envPath != "" {
               path = envPath
           }

           // Step 2: Layer 1 — Defaults (zero-valued Config)
           cfg := DefaultConfig()

           // Step 3: Layer 2 — Global config (~/.m31a/config.toml)
           // Decode global config file; missing is not an error
           var hasGlobal bool
           if path != "" {
               _, err := toml.DecodeFile(path, cfg)
               if err != nil {
                   if !os.IsNotExist(err) {
                       return nil, fmt.Errorf("decode global config %s: %w", path, err)
                   }
               } else {
                   hasGlobal = true
               }
           }

           // Step 4: Layer 3 — Environment variable overrides
           // (existing code, moved inside Load after global decode)
           if theme := os.Getenv("M31A_THEME"); theme != "" {
               cfg.UI.Theme = theme
           }
           if model := os.Getenv("M31A_DEFAULT_MODEL"); model != "" {
               cfg.Model.Default = model
           }

           // Step 5: Layer 4 — Project-level config (m31a.toml in cwd)
           cwd, err := os.Getwd()
           if err == nil {
               if projectPath := findProjectConfig(cwd); projectPath != "" {
                   // Decode project config into a fresh cfg2, then merge non-zero fields
                   // into cfg. This way project config overrides global+env.
                   var projectCfg Config
                   if _, err := toml.DecodeFile(projectPath, &projectCfg); err != nil {
                       // Log a warning but continue — project config is optional
                       slog.Warn("failed to decode project config", "path", projectPath, "error", err)
                   } else {
                       mergeConfig(cfg, &projectCfg)
                   }
               }
           }

           // Step 6: Validate
           if err := validateConfig(cfg); err != nil {
               return nil, fmt.Errorf("config validation: %w", err)
           }

           // Step 7: Variable substitution
           applyVarSubstitution(cfg)

           return cfg, nil
       }
       ```

    ```go
        if cfg.Model.ContextWarningThreshold < 0 || cfg.Model.ContextWarningThreshold > 1 {
            errs = append(errs, ValidationError{
                Field: "model.context_warning_threshold",
    ```

    **Part B — Variable substitution:**

    Add `applyVarSubstitution(cfg *Config)` to `internal/config/loader.go`:

    ```go
    var varRe = regexp.MustCompile(`\$\{([^}]+)\}`)

    func applyVarSubstitution(cfg *Config) {
        // Walk all string fields and replace ${VAR} patterns with env values
        cfg.Provider.Default = substituteVars(cfg.Provider.Default)
        cfg.Provider.OpenRouter.APIKey = substituteVars(cfg.Provider.OpenRouter.APIKey)
        cfg.Provider.Zen.APIKey = substituteVars(cfg.Provider.Zen.APIKey)
        cfg.Model.Default = substituteVars(cfg.Model.Default)
        cfg.UI.Theme = substituteVars(cfg.UI.Theme)
        cfg.Permissions.DefaultMode = substituteVars(cfg.Permissions.DefaultMode)

        // Also substitute in PermissionRule.Pattern and .Tool
        for i := range cfg.Permissions.Rules {
            cfg.Permissions.Rules[i].Tool = substituteVars(cfg.Permissions.Rules[i].Tool)
            cfg.Permissions.Rules[i].Pattern = substituteVars(cfg.Permissions.Rules[i].Pattern)
            cfg.Permissions.Rules[i].Action = substituteVars(cfg.Permissions.Rules[i].Action)
        }
    }

    func substituteVars(s string) string {
        if s == "" || !strings.Contains(s, "${") {
            return s
        }
        return varRe.ReplaceAllStringFunc(s, func(match string) string {
            // Extract var name from ${VAR}
            name := match[2 : len(match)-1]
            if val, ok := os.LookupEnv(name); ok {
                return val
            }
            // Keep ${VAR} as-is if env var not set
            return match
        })
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Config file on disk → Load() | TOML config files read from user-writable directories (~/.m31a/ and cwd) |
| Env var → config | Environment variables read via os.Getenv() for substitution and overrides |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-02-01 | Tampering | m31a.toml in project root | accept | Config files are user-owned; if an attacker can write to project root, M31A is not the attack vector |
| T-11-02-02 | Information Disclosure | Env var expansion logs values | mitigate | Var substitution uses os.LookupEnv; no logging of expanded values. slog.Warn for decode failure only |
| T-11-02-03 | Tampering | Validation error composition | mitigate | validateConfig collects all errors then returns; no partial-apply risk |
| T-11-02-04 | Spoofing | Unknown config keys silently ignored | accept | Forward-compat: unknown keys produce warning but not error; prevents breakage on config upgrade |
</threat_model>
