       ```go
       import "github.com/bmatcuk/doublestar/v4"

       func (d *dispatcher) checkPermission(ctx context.Context, toolName string, input types.ToolInput) (bool, *types.PermissionContext, error) {
           // Step 1: Check rules first (rules take priority)
           for _, rule := range d.rules {
               // Tool filter: if rule.Tool is set, only match that tool name
               if rule.Tool != "" && !matchToolName(rule.Tool, toolName) {
                   continue
               }

               // Pattern filter: if rule.Pattern is set, check if it matches
               // any string value in ToolInput.Params
               if rule.Pattern != "" {
                   matched := matchAnyParamValue(rule.Pattern, input.Params)
                   if !matched {
                       continue
                   }
               }

               // Rule matched
               pctx := &types.PermissionContext{
                   RuleTool:    rule.Tool,
                   RulePattern: rule.Pattern,
                   RuleAction:  rule.Action,
                   Source:      "rule",
               }

               switch rule.Action {
               case "allow":
                   return true, pctx, nil
               case "deny":
                   return false, pctx, types.ErrPermissionDenied
               case "ask":
                   // Fall through to ask user
                   return false, pctx, nil // nil error means "ask"
               }
           }

           // No rule matched — use agent default action or fall through to risk level
           if d.activeAgent != "default" {
               if profile, ok := d.agents[d.activeAgent]; ok && profile.DefaultAction != "" {
                   switch profile.DefaultAction {
                   case "allow":
                       return true, &types.PermissionContext{Source: "agent_default"}, nil
                   case "deny":
                       return false, &types.PermissionContext{Source: "agent_default"}, types.ErrPermissionDenied
                   }
               }
           }

           // No matching rule — fall through to RiskLevel default behavior
           return false, &types.PermissionContext{Source: "risk_level"}, nil
       }

       func matchToolName(pattern, name string) bool {
           if pattern == name || pattern == "*" {
               return true
           }
           matched, _ := doublestar.Match(pattern, name)
           return matched
       }

       func matchAnyParamValue(pattern string, params map[string]any) bool {
           for _, v := range params {
               str, ok := v.(string)
               if !ok {
                   continue
               }
               matched, err := doublestar.Match(pattern, str)
               if err != nil {
                   continue
               }
               if matched {
                   return true
               }
           }
           return false
       }
       ```

    5. Update `execute` method to use `checkPermission`:
       ```go
       func (d *dispatcher) execute(ctx context.Context, call types.ToolCall) (types.ToolResult, error) {
           tool, ok := d.tools[call.Name]
           if !ok {
               return types.ToolResult{}, fmt.Errorf("unknown tool: %s", call.Name)
           }

           input := types.ToolInput{
               Name:   call.Name,
               Params: make(map[string]any),
           }
           if err := json.Unmarshal(call.Input, &input.Params); err != nil {
               return types.ToolResult{}, fmt.Errorf("unmarshal tool input: %w", err)
           }

           // Check rules first
           d.mu.RLock()
           allowed, pctx, err := d.checkPermission(ctx, call.Name, input)
           d.mu.RUnlock()

           if err != nil {
               return types.ToolResult{Error: err.Error()}, nil
           }

           if !allowed {
               if pctx != nil && pctx.Source == "rule" && pctx.RuleAction == "ask" {
                   // Send PermissionRequest, await PermissionResponse
                   // pctx is attached to the request for audit display
                   req := types.PermissionRequest{
                       ToolName:    call.Name,
                       Command:     formatInput(input.Params),
                       RiskLevel:   tool.RiskLevel(),
                       TimeoutSecs: 300,
                   }
                   // PermissionContext flows through request — will be displayed in modal
                   _ = req  // existing permission flow continues

                   // Fall through to RiskLevel check (existing behavior for dangerous tools)
                   if tool.RiskLevel() == types.RiskDangerous || tool.RiskLevel() == types.RiskDestructive {
                       // existing permission flow (no change)
                   } else {
                       return types.ToolResult{}, types.ErrPermissionDenied
                   }
               }
           }

           return tool.Execute(ctx, input)
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Config file rules → Dispatcher | PermissionRule patterns from config.toml control tool execution gating |
| Tool params → doublestar.Match | Untrusted tool input (file paths, command args) flows through glob matcher |
| Permission modal → User | Modal displays matched rule info; user decides allow/deny |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-03-01 | Tampering | Config rules override defaults | mitigate | Rule matching is additive over RiskLevel; deny rules cannot be bypassed by tool behaviour |
| T-11-03-02 | Information Disclosure | doublestar.Match error handling | mitigate | Match errors are silently skipped (continue), no path/pattern leakage |
| T-11-03-03 | Elevation of Privilege | "allow" rule bypasses modal | accept | User explicitly configured the rule; no privilege elevation path |
| T-11-03-04 | Denial of Service | Catastrophic regex via pattern | mitigate | doublestar uses glob syntax (not regex); worst case is O(n) matching |
| T-11-03-05 | Spoofing | Non-string param values | mitigate | matchAnyParamValue type-asserts to string; non-strings are skipped safely |
</threat_model>
