Replace with strings.Builder:
```go
var planCtxBuilder strings.Builder
for _, group := range planSummary.Groups {
    fmt.Fprintf(&planCtxBuilder, "### %s\n", group.Category)

        fmt.Fprintf(&planCtxBuilder, "- [%s] %s: %s\n", change.Action, change.File, change.Description)
    }
    planCtxBuilder.WriteString("\n")
}
```
