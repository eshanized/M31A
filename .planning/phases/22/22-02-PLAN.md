1. Remove hardcoded model maps from both providers
2. Set all capabilities to false by default:
   ```go
   func parseModelCapabilities(modelID string) types.CapFlags {
       // Start with all false
       caps := types.CapFlags{}
       
       // Try to infer from model ID patterns
       // This is heuristic only — API should provide truth
       id := strings.ToLower(modelID)
       if strings.Contains(id, "vision") || strings.Contains(id, "multimodal") {
           caps.Vision = true
       }
       if strings.Contains(id, "reason") || strings.Contains(id, "thinking") {
           caps.Reasoning = true
       }
       // Tools capability is assumed true for all models
       caps.Tools = true
       
       return caps
   }
   ```
